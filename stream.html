// ========================================================================== // ПРОФЕССИОНАЛЬНЫЙ БЕСШОВНЫЙ АУДИОДВИЖОК ДЛЯ STREAM (START ON GESTURE) // ========================================================================== const TARGET_VOLUME = 0.65; const FADE_TIME = 1.8; const CROSSFADE_TIME = 3.0; let audioCtx = null; let masterGain = null; let playerA = null; let playerB = null; let gainA = null; let gainB = null; let isPlaying = false; let userManuallyMuted = false; let crossfadeTimer = null; let activePlayer = 'A'; function getAudioContext() { if (!audioCtx) { const AudioContextClass = window.AudioContext || window.webkitAudioContext; audioCtx = new AudioContextClass(); masterGain = audioCtx.createGain(); masterGain.gain.setValueAtTime(0, audioCtx.currentTime); masterGain.connect(audioCtx.destination); playerA = new Audio('batlicsea.mp3'); playerB = new Audio('batlicsea.mp3'); playerA.preload = 'auto'; playerB.preload = 'auto'; const sourceA = audioCtx.createMediaElementSource(playerA); const sourceB = audioCtx.createMediaElementSource(playerB); gainA = audioCtx.createGain(); gainB = audioCtx.createGain(); sourceA.connect(gainA).connect(masterGain); sourceB.connect(gainB).connect(masterGain); } if (audioCtx.state === 'suspended') { audioCtx.resume(); } return audioCtx; } function scheduleCrossfade(fromPlayer, toPlayer, fromGain, toGain) { if (!isPlaying) return; const duration = fromPlayer.duration; if (!duration || isNaN(duration)) { fromPlayer.addEventListener('loadedmetadata', () => { scheduleCrossfade(fromPlayer, toPlayer, fromGain, toGain); }, { once: true }); return; } const timeLeft = duration - fromPlayer.currentTime; const timeUntilFade = Math.max(0, (timeLeft - CROSSFADE_TIME) * 1000); clearTimeout(crossfadeTimer); crossfadeTimer = setTimeout(() => { if (!isPlaying) return; toPlayer.currentTime = 0; toGain.gain.setValueAtTime(0.001, audioCtx.currentTime); toPlayer.play().then(() => { const now = audioCtx.currentTime; toGain.gain.exponentialRampToValueAtTime(1.0, now + CROSSFADE_TIME); fromGain.gain.setValueAtTime(Math.max(0.001, fromGain.gain.value), now); fromGain.gain.exponentialRampToValueAtTime(0.001, now + CROSSFADE_TIME); activePlayer = activePlayer === 'A' ? 'B' : 'A'; setTimeout(() => { fromPlayer.pause(); fromPlayer.currentTime = 0; if (activePlayer === 'B') { scheduleCrossfade(playerB, playerA, gainB, gainA); } else { scheduleCrossfade(playerA, playerB, gainA, gainB); } }, CROSSFADE_TIME * 1000); }).catch(err => console.warn(err)); }, timeUntilFade); } function playAudio() { if (userManuallyMuted) return; const ctx = getAudioContext(); isPlaying = true; soundBtn.classList.add('playing'); soundBtn.setAttribute('aria-label', 'Выключить звук'); gainA.gain.setValueAtTime(1.0, ctx.currentTime); gainB.gain.setValueAtTime(0.001, ctx.currentTime); activePlayer = 'A'; playerA.play().then(() => { const now = ctx.currentTime; masterGain.gain.cancelScheduledValues(now); masterGain.gain.setValueAtTime(Math.max(0.0001, masterGain.gain.value), now); masterGain.gain.exponentialRampToValueAtTime(TARGET_VOLUME, now + FADE_TIME); scheduleCrossfade(playerA, playerB, gainA, gainB); removeUnlockListeners(); }).catch(() => { attachUnlockListeners(); }); } function pauseAudio() { if (!audioCtx) return; isPlaying = false; userManuallyMuted = true; clearTimeout(crossfadeTimer); soundBtn.classList.remove('playing'); soundBtn.setAttribute('aria-label', 'Включить звук'); removeUnlockListeners(); const now = audioCtx.currentTime; masterGain.gain.cancelScheduledValues(now); masterGain.gain.setValueAtTime(Math.max(0.0001, masterGain.gain.value), now); masterGain.gain.exponentialRampToValueAtTime(0.0001, now + FADE_TIME); setTimeout(() => { if (!isPlaying) { playerA.pause(); playerB.pause(); } }, FADE_TIME * 1000); } function onUserGesture() { if (!userManuallyMuted && !isPlaying) { playAudio(); } } function attachUnlockListeners() { window.addEventListener('mousemove', onUserGesture, { once: true }); window.addEventListener('wheel', onUserGesture, { once: true, passive: true }); window.addEventListener('touchstart', onUserGesture, { once: true, passive: true }); window.addEventListener('pointerdown', onUserGesture, { once: true }); } function removeUnlockListeners() { window.removeEventListener('mousemove', onUserGesture); window.removeEventListener('wheel', onUserGesture); window.removeEventListener('touchstart', onUserGesture); window.removeEventListener('pointerdown', onUserGesture); } // Пробуем стартовать сразу playAudio(); soundBtn.addEventListener('click', (e) => { e.stopPropagation(); if (!isPlaying) { userManuallyMuted = false; playAudio(); } else { pauseAudio(); } });
