// ========================================================================== // БЕСШОВНЫЙ АУДИОДВИЖОК (WEB AUDIO API + CROSSFADE) // ========================================================================== const soundBtn = document.getElementById('soundBtn'); const TARGET_VOLUME = 0.65; const FADE_TIME = 1.8; // Длительность мягкого старта/остановки (сек) const CROSSFADE_TIME = 3.0; // Длина перехлёста конца и начала лупа (сек) let audioCtx = null; let masterGain = null; let playerA = null; let playerB = null; let gainA = null; let gainB = null; let isPlaying = false; let crossfadeTimer = null; let activePlayer = 'A'; // Текущий играющий плеер function getAudioContext() { if (!audioCtx) { const AudioContextClass = window.AudioContext || window.webkitAudioContext; audioCtx = new AudioContextClass(); masterGain = audioCtx.createGain(); masterGain.gain.setValueAtTime(0, audioCtx.currentTime); masterGain.connect(audioCtx.destination); // Создаем два параллельных плеера playerA = new Audio('batlicsea.mp3'); playerB = new Audio('batlicsea.mp3'); playerA.preload = 'auto'; playerB.preload = 'auto'; const sourceA = audioCtx.createMediaElementSource(playerA); const sourceB = audioCtx.createMediaElementSource(playerB); gainA = audioCtx.createGain(); gainB = audioCtx.createGain(); sourceA.connect(gainA).connect(masterGain); sourceB.connect(gainB).connect(masterGain); } if (audioCtx.state === 'suspended') { audioCtx.resume(); } return audioCtx; } // Алгоритм бесшовной сшивки конца трека с началом function scheduleCrossfade(fromPlayer, toPlayer, fromGain, toGain) { if (!isPlaying) return; const duration = fromPlayer.duration; if (!duration || isNaN(duration)) { // Если метаданные еще подгружаются, ждем fromPlayer.addEventListener('loadedmetadata', () => { scheduleCrossfade(fromPlayer, toPlayer, fromGain, toGain); }, { once: true }); return; } const timeLeft = duration - fromPlayer.currentTime; const timeUntilFade = Math.max(0, (timeLeft - CROSSFADE_TIME) * 1000); clearTimeout(crossfadeTimer); crossfadeTimer = setTimeout(() => { if (!isPlaying) return; // 1. Сбрасываем и готовим следующий плеер toPlayer.currentTime = 0; toGain.gain.setValueAtTime(0.001, audioCtx.currentTime); toPlayer.play().then(() => { const now = audioCtx.currentTime; // Плавное нарастание нового трека и угасание старого внахлёст toGain.gain.exponentialRampToValueAtTime(1.0, now + CROSSFADE_TIME); fromGain.gain.setValueAtTime(Math.max(0.001, fromGain.gain.value), now); fromGain.gain.exponentialRampToValueAtTime(0.001, now + CROSSFADE_TIME); // Переключаем активную роль activePlayer = activePlayer === 'A' ? 'B' : 'A'; setTimeout(() => { fromPlayer.pause(); fromPlayer.currentTime = 0; // Планируем следующий круг для нового активного плеера if (activePlayer === 'B') { scheduleCrossfade(playerB, playerA, gainB, gainA); } else { scheduleCrossfade(playerA, playerB, gainA, gainB); } }, CROSSFADE_TIME * 1000); }).catch(err => console.warn(err)); }, timeUntilFade); } function playAudio() { const ctx = getAudioContext(); isPlaying = true; soundBtn.classList.add('playing'); soundBtn.setAttribute('aria-label', 'Выключить звук'); gainA.gain.setValueAtTime(1.0, ctx.currentTime); gainB.gain.setValueAtTime(0.001, ctx.currentTime); activePlayer = 'A'; playerA.play().then(() => { const now = ctx.currentTime; // Мягкий Fade-In общего мастера masterGain.gain.cancelScheduledValues(now); masterGain.gain.setValueAtTime(Math.max(0.0001, masterGain.gain.value), now); masterGain.gain.exponentialRampToValueAtTime(TARGET_VOLUME, now + FADE_TIME); scheduleCrossfade(playerA, playerB, gainA, gainB); }).catch(err => { isPlaying = false; soundBtn.classList.remove('playing'); console.warn("Автоплей заблокирован:", err); }); } function pauseAudio() { if (!audioCtx) return; isPlaying = false; clearTimeout(crossfadeTimer); soundBtn.classList.remove('playing'); soundBtn.setAttribute('aria-label', 'Включить звук'); const now = audioCtx.currentTime; masterGain.gain.cancelScheduledValues(now); masterGain.gain.setValueAtTime(Math.max(0.0001, masterGain.gain.value), now); masterGain.gain.exponentialRampToValueAtTime(0.0001, now + FADE_TIME); setTimeout(() => { if (!isPlaying) { playerA.pause(); playerB.pause(); } }, FADE_TIME * 1000); } soundBtn.addEventListener('click', (e) => { e.stopPropagation(); if (!isPlaying) { playAudio(); } else { pauseAudio(); } }); // Автоскрытие тумблера при неактивности let soundUiTimeout = null; let isHoveringSound = false; function triggerSoundUi() { soundBtn.classList.add('visible'); clearTimeout(soundUiTimeout); if (!isHoveringSound) { soundUiTimeout = setTimeout(() => { soundBtn.classList.remove('visible'); }, 2500); } } soundUiTimeout = setTimeout(() => { if (!isHoveringSound) soundBtn.classList.remove('visible'); }, 2500); window.addEventListener('mousemove', triggerSoundUi); window.addEventListener('touchstart', triggerSoundUi, { passive: true }); window.addEventListener('scroll', triggerSoundUi, { passive: true }); soundBtn.addEventListener('mouseenter', () => { isHoveringSound = true; clearTimeout(soundUiTimeout); soundBtn.classList.add('visible'); }); soundBtn.addEventListener('mouseleave', () => { isHoveringSound = false; triggerSoundUi(); });
